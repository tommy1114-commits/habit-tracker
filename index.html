跨裝置雲端同步設定 (手機與電腦同步)

💡 為什麼電腦和手機的紀錄一開始沒有同步？
瀏覽器出於隱私安全考量，資料預設各自保存在個別手機與電腦中。若要跨裝置自動同步，只需透過免費的 Google 雲端同步，在手機與電腦填入相同網址，打卡或看書計時就會即時自動雙向同步！

🌟 方案 A：Google 個人免費雲端同步 (最推薦 · 永久即時雙向同步)

推薦

資料直接存放在您個人 Google 帳號中，完全免費且安全無隱私疑慮。只要將部屬好的 Google 雲端網址貼在下方，手機與電腦填入同一個網址即可！

👉 點此查看：如何 1 分鐘免費取得您的 Google 同步網址？
  1. 前往 Google Apps Script (點此開啟)，點擊左上角「新專案」。
  2. 刪除原本編輯區的所有內容，複製並貼上下方的 18 行同步代碼： 
    function saveToStorage(contents) {
  var props = PropertiesService.getUserProperties();
  var chunkSize = 8000;
  var count = Math.ceil(contents.length / chunkSize);
  var oldChunks = parseInt(props.getProperty('habit_chunks_count') || '0', 10);
  for (var i = 0; i < oldChunks; i++) {
    props.deleteProperty('habit_chunk_' + i);
  }
  if (count <= 1) {
    props.setProperty('habit_tracker_data', contents);
    props.deleteProperty('habit_chunks_count');
  } else {
    props.setProperty('habit_chunks_count', count.toString());
    for (var j = 0; j < count; j++) {
      props.setProperty('habit_chunk_' + j, contents.substr(j * chunkSize, chunkSize));
    }
    props.deleteProperty('habit_tracker_data');
  }
}

function loadFromStorage() {
  var props = PropertiesService.getUserProperties();
  var countStr = props.getProperty('habit_chunks_count');
  if (countStr) {
    var count = parseInt(countStr, 10);
    var result = '';
    for (var i = 0; i < count; i++) {
      result += (props.getProperty('habit_chunk_' + i) || '');
    }
    return result;
  }
  return props.getProperty('habit_tracker_data') || '';
}

function doGet(e) {
  try {
    var data = loadFromStorage();
    var output = data ? data : JSON.stringify({ empty: true });
    return ContentService.createTextOutput(output)
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var contents = e.postData.contents;
    if (!contents) {
      return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'No content' }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    saveToStorage(contents);
    return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
